let improve it
@tool
@desc "use this to delete a user"
function delete_person(id: string, isAdmin: bool): string {
    if not isAdmin {
        return "user is not an admin"
    }

    let result = delete_user(id)

    if result.error.isEmpty() {
        return "user deleted successfully"
    }

    return result.error
}
