## 1. Normalized Message Format (The Foundation)

First, we need a **provider-agnostic message format** that handles reasoning/thinking content uniformly:

## 2. Middleware Architecture (Beyond LangChain)

LangChain v1.0 has 6 hooks but they're limited. Here's **Auwgent's comprehensive middleware system**:

```typescript
namespace Auwgent.Middleware {
  
  // Context passed through middleware chain
  export interface MiddlewareContext {
    agentName: string;
    runId: string;
    attempt: number;        // For retries
    startTime: number;
    
    // Mutable state (middleware can modify)
    state: Record<string, any>;
    
    // Immutable request/response (use modify methods)
    readonly originalRequest: Messages.ModelRequest;
    readonly originalInput: any;
    
    // Current working copies
    request: Messages.ModelRequest;
    response?: Messages.ModelResponse;
    
    // Execution metadata
    history: ExecutionEvent[];
    
    // User context (lifecycle hooks)
    userContext?: any;
  }

  export interface ExecutionEvent {
    type: 'model_call' | 'tool_call' | 'thinking' | 'error' | 'checkpoint';
    timestamp: number;
    duration?: number;
    data: any;
  }

  // The 8 middleware hooks (more granular than LangChain's 6)
  export interface AgentMiddleware {
    name: string;
    priority?: number;      // Execution order (lower = earlier)
    
    // 1. Before agent starts (LangChain: before_agent)
    onAgentStart?: (
      context: MiddlewareContext
    ) => Promise<void> | void;

    // 2. Before model call - modify prompt/messages (LangChain: before_model)
    onBeforeModel?: (
      context: MiddlewareContext
    ) => Promise<Messages.ModelRequest | void> | Messages.ModelRequest | void;

    // 3. Wrap the actual model call (NEW - LangChain has this but limited)
    wrapModelCall?: (
      context: MiddlewareContext,
      execute: () => Promise<Messages.ModelResponse>
    ) => Promise<Messages.ModelResponse>;

    // 4. Thinking/Reasoning hook (NEW - LangChain doesn't have this)
    onThinking?: (
      context: MiddlewareContext,
      thinking: Messages.ThinkingBlock
    ) => Promise<Messages.ThinkingBlock | void> | Messages.ThinkingBlock | void;

    // 5. After model response, before tool execution (LangChain: after_model)
    onAfterModel?: (
      context: MiddlewareContext,
      response: Messages.ModelResponse
    ) => Promise<Messages.ModelResponse | void> | Messages.ModelResponse | void;

    // 6. Before tool execution (NEW - LangChain: wrap_tool_call but no pre-hook)
    onBeforeTool?: (
      context: MiddlewareContext,
      toolCall: Messages.ToolUseBlock
    ) => Promise<boolean | void> | boolean | void;  // Return false to block

    // 7. Wrap tool execution (LangChain: wrap_tool_call)
    wrapToolCall?: (
      context: MiddlewareContext,
      toolCall: Messages.ToolUseBlock,
      execute: (input: any) => Promise<any>
    ) => Promise<any>;

    // 8. After tool execution (NEW)
    onAfterTool?: (
      context: MiddlewareContext,
      toolCall: Messages.ToolUseBlock,
      result: any
    ) => Promise<void> | void;

    // 9. On agent completion/error (LangChain: after_agent)
    onAgentEnd?: (
      context: MiddlewareContext,
      result: any,
      error?: Error
    ) => Promise<void> | void;

    // 10. Error handling (NEW - granular error recovery)
    onError?: (
      context: MiddlewareContext,
      error: Error,
      phase: 'model' | 'tool' | 'thinking'
    ) => Promise<{ retry: boolean; delay?: number } | void> | { retry: boolean; delay?: number } | void;
  }

  // Middleware registry and chain execution
  export class MiddlewareChain {
    private middlewares: AgentMiddleware[] = [];
    
    register(middleware: AgentMiddleware): void {
      this.middlewares.push(middleware);
      this.middlewares.sort((a, b) => (a.priority || 0) - (b.priority || 0));
    }
    
    async execute(
      initialContext: MiddlewareContext,
      modelExecutor: () => Promise<Messages.ModelResponse>,
      toolExecutor: (call: Messages.ToolUseBlock) => Promise<any>
    ): Promise<{ context: MiddlewareContext; result: any }> {
      const context = { ...initialContext, history: [] };
      
      try {
        // 1. Agent start
        for (const mw of this.middlewares) {
          if (mw.onAgentStart) await mw.onAgentStart(context);
        }
        
        // 2. Before model
        for (const mw of this.middlewares) {
          if (mw.onBeforeModel) {
            const modified = await mw.onBeforeModel(context);
            if (modified) context.request = modified;
          }
        }
        
        // 3. Wrap model call
        let response: Messages.ModelResponse;
        const modelWrapper = this.buildModelWrapper(modelExecutor);
        response = await modelWrapper(context);
        context.response = response;
        
        // 4. Process thinking blocks if present
        for (const block of response.content) {
          if (block.type === 'thinking') {
            for (const mw of this.middlewares) {
              if (mw.onThinking) {
                const modified = await mw.onThinking(context, block);
                if (modified) Object.assign(block, modified);
              }
            }
          }
        }
        
        // 5. After model
        for (const mw of [...this.middlewares].reverse()) {
          if (mw.onAfterModel) {
            const modified = await mw.onAfterModel(context, response);
            if (modified) context.response = modified;
          }
        }
        
        // 6. & 7. & 8. Handle tool calls
        for (const block of response.content) {
          if (block.type === 'tool_use') {
            const shouldExecute = await this.runBeforeToolHooks(context, block);
            if (shouldExecute) {
              const result = await this.runToolWithWrappers(context, block, toolExecutor);
              await this.runAfterToolHooks(context, block, result);
            }
          }
        }
        
        // 9. Agent end
        for (const mw of [...this.middlewares].reverse()) {
          if (mw.onAgentEnd) await mw.onAgentEnd(context, context.response);
        }
        
        return { context, result: context.response };
        
      } catch (error) {
        // 10. Error handling
        const handled = await this.handleError(context, error as Error);
        if (!handled.retry) throw error;
        
        // Retry logic
        await new Promise(r => setTimeout(r, handled.delay || 1000));
        return this.execute(initialContext, modelExecutor, toolExecutor);
      }
    }
    
    private buildModelWrapper(
      executor: () => Promise<Messages.ModelResponse>
    ): (ctx: MiddlewareContext) => Promise<Messages.ModelResponse> {
      return this.middlewares
        .filter(mw => mw.wrapModelCall)
        .reduceRight(
          (next, mw) => (ctx) => mw.wrapModelCall!(ctx, next),
          executor
        );
    }
    
    private async runBeforeToolHooks(
      context: MiddlewareContext,
      toolCall: Messages.ToolUseBlock
    ): Promise<boolean> {
      for (const mw of this.middlewares) {
        if (mw.onBeforeTool) {
          const shouldContinue = await mw.onBeforeTool(context, toolCall);
          if (shouldContinue === false) return false;
        }
      }
      return true;
    }
    
    private async runToolWithWrappers(
      context: MiddlewareContext,
      toolCall: Messages.ToolUseBlock,
      executor: (input: any) => Promise<any>
    ): Promise<any> {
      const wrappers = this.middlewares.filter(mw => mw.wrapToolCall);
      
      let wrapped = executor;
      for (const mw of wrappers.reverse()) {
        const current = wrapped;
        wrapped = (input: any) => mw.wrapToolCall!(context, toolCall, () => current(input));
      }
      
      return wrapped(toolCall.input);
    }
    
    private async runAfterToolHooks(
      context: MiddlewareContext,
      toolCall: Messages.ToolUseBlock,
      result: any
    ): Promise<void> {
      for (const mw of [...this.middlewares]) {
        if (mw.onAfterTool) await mw.onAfterTool(context, toolCall, result);
      }
    }
    
    private async handleError(
      context: MiddlewareContext,
      error: Error
    ): Promise<{ retry: boolean; delay?: number }> {
      // Phase detection based on error type
      const phase = error.name === 'ToolExecutionError' ? 'tool' : 
                    error.name === 'ThinkingError' ? 'thinking' : 'model';
      
      for (const mw of [...this.middlewares]) {
        if (mw.onError) {
          const decision = await mw.onError(context, error, phase);
          if (decision) return decision;
        }
      }
      return { retry: false };
    }
  }
}
```

```typescript
namespace Auwgent.Middleware.Builtins {
  
  // 1. Reasoning Token Handler (NEW - LangChain doesn't handle this)
  export class ReasoningMiddleware implements AgentMiddleware {
    name = 'reasoning_handler';
    priority = 50;  // Run early
    
    async onThinking(
      context: MiddlewareContext,
      thinking: Messages.ThinkingBlock
    ): Promise<Messages.ThinkingBlock> {
      // Log reasoning for debugging even if provider hides it
      if (!thinking.isRedacted) {
        context.history.push({
          type: 'thinking',
          timestamp: Date.now(),
          data: thinking.thinking
        });
      }
      
      // Optionally modify thinking presentation
      if (thinking.summary) {
        // Anthropic provides summary, use it
        return thinking;
      }
      
      // For OpenAI o1 which hides thinking, create summary from token count
      if (thinking.isRedacted && thinking.tokenCount) {
        thinking.summary = `(${thinking.tokenCount} reasoning tokens used)`;
      }
      
      return thinking;
    }
    
    onAfterModel(context: MiddlewareContext): void {
      // Track reasoning token usage
      const usage = context.response?.usage;
      if (usage?.reasoning) {
        context.state.reasoningTokensTotal = 
          (context.state.reasoningTokensTotal || 0) + usage.reasoning;
      }
    }
  }

  // 2. Provider Normalization Middleware (NEW)
  export class ProviderNormalizationMiddleware implements AgentMiddleware {
    name = 'provider_normalizer';
    priority = 10;  // Run first
    
    async onBeforeModel(
      context: MiddlewareContext
    ): Promise<Messages.ModelRequest> {
      const req = context.request;
      
      // Normalize reasoning config for specific provider
      const normalized = { ...req };
      const provider = this.detectProvider(req.model);
      
      if (provider === 'anthropic' && req.reasoning?.enabled) {
        // Convert to Anthropic's extended thinking format
        normalized.extraBody = {
          thinking: {
            type: 'enabled',
            budget_tokens: req.reasoning.budgetTokens || 16000
          }
        };
        delete normalized.reasoning;  // Remove from standard fields
      }
      
      if (provider === 'openai' && req.reasoning?.enabled) {
        // OpenAI uses reasoning_effort
        normalized.reasoning_effort = req.reasoning.effort || 'medium';
        delete (normalized as any).reasoning;
      }
      
      return normalized;
    }
    
    async onAfterModel(
      context: MiddlewareContext,
      response: Messages.ModelResponse
    ): Promise<Messages.ModelResponse> {
      // Normalize response back to standard format
      const normalized = { ...response };
      
      // Extract thinking blocks from various provider formats
      normalized.content = this.extractThinkingBlocks(response);
      
      return normalized;
    }
    
    private detectProvider(model: string): string {
      if (model.includes('claude')) return 'anthropic';
      if (model.includes('gpt') || model.includes('o1') || model.includes('o3')) return 'openai';
      if (model.includes('gemini')) return 'google';
      return 'unknown';
    }
    
    private extractThinkingBlocks(response: Messages.ModelResponse): Messages.ContentBlock[] {
      const content = response.content;
      const blocks: Messages.ContentBlock[] = [];
      
      for (const block of content) {
        if (typeof block === 'string') {
          blocks.push({ type: 'text', text: block });
          continue;
        }
        
        // Anthropic thinking blocks
        if (block.type === 'thinking' || (block as any).thinking) {
          blocks.push({
            type: 'thinking',
            thinking: (block as any).thinking || '',
            signature: (block as any).signature,
            tokenCount: (block as any).thinking_tokens
          });
        }
        // OpenAI reasoning content
        else if ((block as any).reasoning_content) {
          blocks.push({
            type: 'thinking',
            thinking: (block as any).reasoning_content,
            isRedacted: false
          });
        }
        else {
          blocks.push(block as Messages.ContentBlock);
        }
      }
      
      return blocks;
    }
  }

  // 3. Rate Limiting Middleware (improved over LangChain)
  export class RateLimitMiddleware implements AgentMiddleware {
    name = 'rate_limiter';
    priority = 20;
    
    private tokens: number;
    private lastRefill: number;
    private readonly maxTokens: number;
    private readonly refillRate: number;
    
    constructor(config: { maxRequests: number; windowMs: number }) {
      this.maxTokens = config.maxRequests;
      this.tokens = config.maxRequests;
      this.refillRate = config.windowMs;
      this.lastRefill = Date.now();
    }
    
    async onBeforeModel(context: MiddlewareContext): Promise<void> {
      this.refillTokens();
      
      if (this.tokens < 1) {
        const waitTime = this.calculateWaitTime();
        throw new RateLimitError(`Rate limit exceeded. Retry after ${waitTime}ms`);
      }
      
      this.tokens--;
    }
    
    private refillTokens(): void {
      const now = Date.now();
      const timePassed = now - this.lastRefill;
      const tokensToAdd = Math.floor(timePassed / this.refillRate) * this.maxTokens;
      
      this.tokens = Math.min(this.maxTokens, this.tokens + tokensToAdd);
      this.lastRefill = now;
    }
    
    private calculateWaitTime(): number {
      return this.refillRate - (Date.now() - this.lastRefill);
    }
  }

  // 4. Human-in-the-Loop (HITL) Middleware
  export class HITLMiddleware implements AgentMiddleware {
    name = 'human_in_the_loop';
    priority = 100;  // Run late (high priority number = later)
    
    constructor(
      private config: {
        shouldInterrupt: (context: MiddlewareContext, toolCall: Messages.ToolUseBlock) => boolean;
        onInterrupt: (context: MiddlewareContext, toolCall: Messages.ToolUseBlock) => Promise<{
          approved: boolean;
          modifiedArgs?: any;
          rejectionReason?: string;
        }>;
      }
    ) {}
    
    async onBeforeTool(
      context: MiddlewareContext,
      toolCall: Messages.ToolUseBlock
    ): Promise<boolean> {
      if (!this.config.shouldInterrupt(context, toolCall)) {
        return true;  // Continue execution
      }
      
      const decision = await this.config.onInterrupt(context, toolCall);
      
      if (!decision.approved) {
        throw new ToolRejectionError(
          decision.rejectionReason || 'Tool execution rejected by human'
        );
      }
      
      if (decision.modifiedArgs) {
        toolCall.input = decision.modifiedArgs;
      }
      
      return true;
    }
  }

  // 5. Retry Middleware with Exponential Backoff
  export class RetryMiddleware implements AgentMiddleware {
    name = 'retry_handler';
    priority = 5;
    
    constructor(private maxRetries: number = 3) {}
    
    async onError(
      context: MiddlewareContext,
      error: Error,
      phase: 'model' | 'tool' | 'thinking'
    ): Promise<{ retry: boolean; delay?: number }> {
      if (context.attempt >= this.maxRetries) {
        return { retry: false };
      }
      
      // Don't retry certain errors
      if (error instanceof AuthenticationError || 
          error instanceof ValidationError) {
        return { retry: false };
      }
      
      const delay = Math.pow(2, context.attempt) * 1000 + Math.random() * 100;
      return { retry: true, delay };
    }
  }

  // 6. Thinking Budget Monitor (NEW - for reasoning models)
  export class ThinkingBudgetMiddleware implements AgentMiddleware {
    name = 'thinking_budget';
    priority = 30;
    
    constructor(private maxThinkingTokens: number) {}
    
    async onBeforeModel(context: MiddlewareContext): Promise<void> {
      const currentUsage = context.state.reasoningTokensTotal || 0;
      const remaining = this.maxThinkingTokens - currentUsage;
      
      if (remaining <= 0) {
        // Disable thinking for this request
        (context.request as any).reasoning = { enabled: false };
        console.warn('Thinking budget exhausted, disabling for this request');
      }
    }
    
    onAfterModel(context: MiddlewareContext): void {
      const usage = context.response?.usage?.reasoning || 0;
      const total = (context.state.reasoningTokensTotal || 0) + usage;
      
      if (total > this.maxThinkingTokens * 0.8) {
        console.warn(`Thinking budget at 80%: ${total}/${this.maxThinkingTokens}`);
      }
    }
  }
}
```

## 4. Driver Interface with Thinking Support

```typescript
namespace Auwgent.Drivers {
  
  export interface ModelDriver {
    readonly name: string;
    readonly supportsThinking: boolean;
    
    // Main execution methods
    complete(request: Messages.ModelRequest): Promise<Messages.ModelResponse>;
    stream(request: Messages.ModelRequest): AsyncGenerator<Messages.StreamChunk>;
    
    // NEW: Thinking/reasoning specific
    getThinkingConfig?(): {
      maxBudgetTokens: number;
      minBudgetTokens: number;
      supportedEffortLevels?: string[];
    };
    
    // Normalize provider-specific formats
    normalizeRequest(request: Messages.ModelRequest): any;
    normalizeResponse(raw: any): Messages.ModelResponse;
  }

  // Anthropic Driver with Extended Thinking
  export class AnthropicDriver implements ModelDriver {
    name = 'anthropic';
    supportsThinking = true;
    
    async complete(request: Messages.ModelRequest): Promise<Messages.ModelResponse> {
      const normalized = this.normalizeRequest(request);
      
      const response = await fetch('https://api.anthropic.com/v1/messages', {
        method: 'POST',
        headers: {
          'x-api-key': this.apiKey,
          'content-type': 'application/json',
          'anthropic-version': '2023-06-01'
        },
        body: JSON.stringify(normalized)
      });
      
      const raw = await response.json();
      return this.normalizeResponse(raw);
    }
    
    normalizeRequest(request: Messages.ModelRequest): any {
      const body: any = {
        model: request.model,
        messages: this.convertMessages(request.messages),
        max_tokens: request.maxTokens || 4096
      };
      
      // Handle extended thinking
      if (request.reasoning?.enabled) {
        body.thinking = {
          type: 'enabled',
          budget_tokens: request.reasoning.budgetTokens || 16000
        };
      }
      
      return body;
    }
    
    normalizeResponse(raw: any): Messages.ModelResponse {
      const content: Messages.ContentBlock[] = [];
      
      for (const block of raw.content) {
        if (block.type === 'thinking') {
          content.push({
            type: 'thinking',
            thinking: block.thinking,
            signature: block.signature,
            tokenCount: raw.usage?.thinking_tokens
          });
        } else if (block.type === 'text') {
          content.push({ type: 'text', text: block.text });
        } else if (block.type === 'tool_use') {
          content.push({
            type: 'tool_use',
            id: block.id,
            name: block.name,
            input: block.input
          });
        }
      }
      
      return {
        id: raw.id,
        model: raw.model,
        provider: 'anthropic',
        stopReason: raw.stop_reason,
        content,
        usage: {
          input: raw.usage?.input_tokens,
          output: raw.usage?.output_tokens,
          reasoning: raw.usage?.thinking_tokens,
          total: raw.usage?.input_tokens + raw.usage?.output_tokens
        },
        isComplete: true
      };
    }
    
    getThinkingConfig() {
      return {
        maxBudgetTokens: 128000,
        minBudgetTokens: 1024,
        supportedEffortLevels: undefined  // Anthropic uses budget_tokens
      };
    }
    
    private convertMessages(messages: Messages.Message[]): any[] {
      // Convert normalized messages to Anthropic format
      return messages.map(m => ({
        role: m.role === 'system' ? 'system' : m.role,
        content: typeof m.content === 'string' ? m.content : 
          m.content.map(c => {
            if (c.type === 'thinking') return { type: 'thinking', thinking: c.thinking };
            if (c.type === 'text') return { type: 'text', text: c.text };
            return c;
          })
      }));
    }
  }

  // OpenAI Driver with o1/o3 Reasoning
  export class OpenAIDriver implements ModelDriver {
    name = 'openai';
    supportsThinking = true;
    
    normalizeRequest(request: Messages.ModelRequest): any {
      const body: any = {
        model: request.model,
        messages: this.convertMessages(request.messages)
      };
      
      // o1/o3 use reasoning_effort
      if (request.reasoning?.enabled && this.isReasoningModel(request.model)) {
        body.reasoning_effort = request.reasoning.effort || 'medium';
        // Reasoning models don't support temperature
        delete body.temperature;
      } else {
        body.temperature = request.temperature;
      }
      
      return body;
    }
    
    normalizeResponse(raw: any): Messages.ModelResponse {
      const message = raw.choices?.[0]?.message;
      
      // Extract reasoning content if present (o1 sometimes exposes this)
      const content: Messages.ContentBlock[] = [];
      
      if (message?.reasoning_content) {
        content.push({
          type: 'thinking',
          thinking: message.reasoning_content,
          isRedacted: false  // Sometimes o1 exposes it
        });
      }
      
      if (message?.content) {
        content.push({ type: 'text', text: message.content });
      }
      
      return {
        id: raw.id,
        model: raw.model,
        provider: 'openai',
        stopReason: raw.choices?.[0]?.finish_reason,
        content,
        usage: {
          input: raw.usage?.prompt_tokens,
          output: raw.usage?.completion_tokens,
          reasoning: raw.usage?.reasoning_tokens,  // May be hidden
          total: raw.usage?.total_tokens
        },
        isComplete: true,
        raw  // Keep raw for debugging
      };
    }
    
    private isReasoningModel(model: string): boolean {
      return model.startsWith('o1') || model.startsWith('o3');
    }
    
    getThinkingConfig() {
      return {
        maxBudgetTokens: 100000,  // o1 can use up to 100k reasoning tokens
        minBudgetTokens: 0,
        supportedEffortLevels: ['low', 'medium', 'high']
      };
    }
  }
}
```

## 5. Usage in Auwgent Agent

```typescript
// User code
import { createResearcher } from './researcher.agent.types';
import { ReasoningMiddleware, RateLimitMiddleware, HITLMiddleware } from 'auwgent/middleware';

const agent = createResearcher({
  tools: { search, analyze },
  
  // Middleware chain configuration
  middleware: [
    new ReasoningMiddleware(),  // Handle thinking content
    new RateLimitMiddleware({ maxRequests: 60, windowMs: 60000 }),
    new HITLMiddleware({
      shouldInterrupt: (ctx, tool) => tool.name === 'delete_data',
      onInterrupt: async (ctx, tool) => {
        // Show UI prompt, wait for human
        return await showApprovalDialog(tool);
      }
    })
  ],
  
  // Reasoning configuration
  reasoning: {
    enabled: true,
    budgetTokens: 32000,  // For Claude 3.7 extended thinking
    effort: 'high',       // For OpenAI o1
    visible: true         // Show thinking to user
  }
});

// Usage with full observability
const result = await agent.run({ query: "Complex research task" });

// Access thinking content
for (const block of result.content) {
  if (block.type === 'thinking') {
    console.log('Thinking process:', block.thinking || block.summary);
    console.log('Tokens used:', block.tokenCount);
  }
}
```
